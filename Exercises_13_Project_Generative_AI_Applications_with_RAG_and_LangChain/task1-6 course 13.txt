from langchain.text_splitter import RecursiveCharacterTextSplitter
latex_text = """\documentclass{article}\begin{document}\maketitle\section{Introduction}
Large language models (LLMs) are a type of machine learning model that can be trained on vast amounts of text data to generate human-like language. In recent years, LLMs have made significant advances in various natural language processing tasks, including language translation, text generation, and sentiment analysis.
\subsection{History of LLMs}
The earliest LLMs were developed in the 1980s and 1990s, but they were limited by the amount of data that could be processed and the computational power available at the time. In the past decade, however, advances in hardware and software have made it possible to train LLMs on massive datasets, leading to significant improvements in performance.
\subsection{Applications of LLMs}
LLMs have many applications in the industry, including chatbots, content creation, and virtual assistants. They can also be used in academia for research in linguistics, psychology, and computational linguistics.
\end{document}
"""
splitter = RecursiveCharacterTextSplitter(chunk_size=300, chunk_overlap=50)
splits = splitter.split_text(latex_text)
print("\n--- Gesplittete Chunks ---\n")
for i, chunk in enumerate(splits):
    print(f"Chunk {i+1}:\n{chunk}\n")

class WatsonxEmbedding:
    def __init__(self):
        # Modellname nur symbolisch
        self.model_id = "mistralai/mixtral-8x7b-instruct-v01"

    def embed_query(self, query: str):
        return [len(query) * 0.01 + i * 0.001 for i in range(384)]

    def embed_documents(self, texts: list[str]) -> list[list[float]]:
        return [[len(text) * 0.01 + i * 0.001 for i in range(384)] for text in texts]

from langchain_community.document_loaders import PyPDFLoader
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_community.vectorstores import Chroma
from rag.watsonx_embedding import WatsonxEmbedding
from langchain.chains import RetrievalQA
from langchain_core.language_models import FakeListLLM  # Dummy-Modell ersetzen bei Bedarf
loader = PyPDFLoader("1.pdf")
docs = loader.load()

# 2. Splitten
splitter = RecursiveCharacterTextSplitter(chunk_size=1000, chunk_overlap=200)
splits = splitter.split_documents(docs)

# 3. Embedding + Vektordatenbank
embedding_model = WatsonxEmbedding()
vectordb = Chroma.from_documents(splits, embedding_model, persist_directory="./chroma_db_task6")
vectordb.persist()

# 4. Retriever definieren
retriever = vectordb.as_retriever()

# 5. Dummy LLM (für echte Fälle LLM wie OpenAI, IBM oder lokal einbinden)
class DummyLLM(FakeListLLM):
    def generate(self, prompts, stop=None):
        return [f"Simulierte Antwort für Prompt: {prompt}" for prompt in prompts]

llm = DummyLLM(responses=["This paper discusses large language models and their adaptation strategies."])

# 6. QA-Chain aufbauen
qa_chain = RetrievalQA.from_chain_type(llm=llm, retriever=retriever)

# 7. Query ausführen
query = "What this paper is talking about?"
result = qa_chain.run(query)

# 8. Ausgabe
print(f"Query: {query}")
print(f"Answer: {result}")
